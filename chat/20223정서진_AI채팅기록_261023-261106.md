```python
import streamlit as st
import pandas as pd
import plotly.express as px

# ----------------------------
# 기본 설정
# ----------------------------
st.set_page_config(page_title="Fitbit 일일 활동 데이터 시각화", layout="wide")
st.title("🏃 Fitbit 일일 활동 데이터 시각화")

DATA_URL = "https://raw.githubusercontent.com/dg2025825-arch/yeppi/refs/heads/main/data/dailyActivity_merged.csv"


# ----------------------------
# 데이터 불러오기
# ----------------------------
@st.cache_data
def load_data(url):
    df = pd.read_csv(url)
    df["ActivityDate"] = pd.to_datetime(df["ActivityDate"], format="%m/%d/%Y", errors="coerce")
    df["Id"] = df["Id"].astype(str)
    return df


try:
    df = load_data(DATA_URL)
except Exception as e:
    st.error(f"데이터를 불러오지 못했습니다: {e}")
    st.stop()

# ----------------------------
# 사이드바 필터
# ----------------------------
st.sidebar.header("🔎 필터")

user_ids = sorted(df["Id"].unique())
selected_ids = st.sidebar.multiselect(
    "사용자 ID 선택 (비우면 전체)", user_ids, default=[]
)

min_date = df["ActivityDate"].min().date()
max_date = df["ActivityDate"].max().date()
date_range = st.sidebar.date_input(
    "날짜 범위", value=(min_date, max_date), min_value=min_date, max_value=max_date
)

filtered = df.copy()
if selected_ids:
    filtered = filtered[filtered["Id"].isin(selected_ids)]
if isinstance(date_range, tuple) and len(date_range) == 2:
    start, end = date_range
    filtered = filtered[
        (filtered["ActivityDate"] >= pd.to_datetime(start))
        & (filtered["ActivityDate"] <= pd.to_datetime(end))
    ]

if filtered.empty:
    st.warning("선택한 조건에 해당하는 데이터가 없습니다.")
    st.stop()

# ----------------------------
# 요약 지표
# ----------------------------
st.subheader("📌 요약 지표")
c1, c2, c3, c4 = st.columns(4)
c1.metric("사용자 수", filtered["Id"].nunique())
c2.metric("평균 걸음 수", f"{filtered['TotalSteps'].mean():,.0f}")
c3.metric("평균 이동 거리(km)", f"{filtered['TotalDistance'].mean():.2f}")
c4.metric("평균 소모 칼로리", f"{filtered['Calories'].mean():,.0f}")

# ----------------------------
# 탭 구성
# ----------------------------
tab1, tab2, tab3, tab4, tab5 = st.tabs(
    ["📋 데이터", "📈 시간별 추이", "🔗 관계 분석", "📊 분포", "⏱️ 활동 시간 비교"]
)

# 탭1: 데이터 확인
with tab1:
    st.write("**데이터 미리보기**")
    st.dataframe(filtered, use_container_width=True)
    st.write("**기초 통계량**")
    st.dataframe(filtered.describe(), use_container_width=True)

# 탭2: 날짜별 추이
with tab2:
    metric = st.selectbox(
        "확인할 항목",
        ["TotalSteps", "TotalDistance", "Calories", "SedentaryMinutes", "VeryActiveMinutes"],
    )
    daily = filtered.groupby("ActivityDate", as_index=False)[metric].mean()
    fig = px.line(daily, x="ActivityDate", y=metric, markers=True,
                  title=f"날짜별 평균 {metric}")
    st.plotly_chart(fig, use_container_width=True)

# 탭3: 관계 분석
with tab3:
    num_cols = filtered.select_dtypes("number").columns.tolist()
    colx, coly = st.columns(2)
    x_axis = colx.selectbox("X축", num_cols, index=num_cols.index("TotalSteps"))
    y_axis = coly.selectbox("Y축", num_cols, index=num_cols.index("Calories"))
    fig = px.scatter(filtered, x=x_axis, y=y_axis, color="Id",
                     opacity=0.7, title=f"{x_axis} vs {y_axis}")
    st.plotly_chart(fig, use_container_width=True)

    st.write("**상관관계 히트맵**")
    corr = filtered[num_cols].corr()
    fig_corr = px.imshow(corr, text_auto=".2f", color_continuous_scale="RdBu_r",
                         aspect="auto")
    st.plotly_chart(fig_corr, use_container_width=True)

# 탭4: 분포
with tab4:
    dist_col = st.selectbox(
        "분포를 볼 항목", ["TotalSteps", "Calories", "TotalDistance", "SedentaryMinutes"]
    )
    fig_hist = px.histogram(filtered, x=dist_col, nbins=30, title=f"{dist_col} 히스토그램")
    st.plotly_chart(fig_hist, use_container_width=True)

    fig_box = px.box(filtered, x="Id", y=dist_col, title=f"사용자별 {dist_col} 박스플롯")
    st.plotly_chart(fig_box, use_container_width=True)

# 탭5: 활동 강도별 시간 비교
with tab5:
    minute_cols = ["VeryActiveMinutes", "FairlyActiveMinutes",
                   "LightlyActiveMinutes", "SedentaryMinutes"]
    avg_minutes = filtered[minute_cols].mean().reset_index()
    avg_minutes.columns = ["활동 강도", "평균 시간(분)"]

    col_a, col_b = st.columns(2)
    with col_a:
        fig_bar = px.bar(avg_minutes, x="활동 강도", y="평균 시간(분)",
                         color="활동 강도", title="활동 강도별 평균 시간")
        st.plotly_chart(fig_bar, use_container_width=True)
    with col_b:
        fig_pie = px.pie(avg_minutes, names="활동 강도", values="평균 시간(분)",
                         title="활동 강도별 비율")
        st.plotly_chart(fig_pie, use_container_width=True)
```

```python
import streamlit as st
import pandas as pd
import plotly.express as px

# ----------------------------
# 기본 설정
# ----------------------------
st.set_page_config(page_title="일일 활동 패턴 → 다음 날 수면 부족 예측?", layout="wide")
st.title("🏃 일일 활동 패턴 시각화")
st.caption("연구 질문: 하루의 활동 패턴만으로 다음 날 수면 부족 여부를 예측할 수 있는가?")

ACTIVITY_URL = "https://raw.githubusercontent.com/dg2025825-arch/yeppi/refs/heads/main/data/dailyActivity_merged.csv"

WEEK_ORDER = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"]
MINUTE_COLS = ["VeryActiveMinutes", "FairlyActiveMinutes", "LightlyActiveMinutes", "SedentaryMinutes"]
REQUIRED_COLS = ["Id", "ActivityDate", "TotalSteps"] + MINUTE_COLS


# ----------------------------
# 데이터 불러오기
# ----------------------------
@st.cache_data
def load_activity(url):
    df = pd.read_csv(url)
    missing = [c for c in REQUIRED_COLS if c not in df.columns]
    if missing:
        raise ValueError(f"필요한 열이 없습니다: {missing}")

    df["ActivityDate"] = pd.to_datetime(df["ActivityDate"], format="%m/%d/%Y", errors="coerce")
    df["Id"] = df["Id"].astype(str)

    # 파생 변수
    df["ActiveMinutes"] = (
        df["VeryActiveMinutes"] + df["FairlyActiveMinutes"] + df["LightlyActiveMinutes"]
    )
    df["ModerateVigorousMinutes"] = df["VeryActiveMinutes"] + df["FairlyActiveMinutes"]
    df["TotalRecordedMinutes"] = df[MINUTE_COLS].sum(axis=1)  # 하루 1440분이면 온전히 기록
    df["Weekday"] = df["ActivityDate"].dt.day_name()
    df["NextDate"] = df["ActivityDate"] + pd.Timedelta(days=1)  # 수면 데이터와 합칠 때 사용
    return df


@st.cache_data
def prepare_sleep(file):
    sl = pd.read_csv(file)
    sl["Id"] = sl["Id"].astype(str)
    sl["SleepDate"] = pd.to_datetime(sl["SleepDay"], errors="coerce").dt.normalize()
    return sl


try:
    df = load_activity(ACTIVITY_URL)
except Exception as e:
    st.error(f"데이터를 불러오지 못했습니다: {e}")
    st.stop()

# ----------------------------
# 사이드바 필터
# ----------------------------
st.sidebar.header("🔎 필터")

user_ids = sorted(df["Id"].unique())
selected_ids = st.sidebar.multiselect("사용자 ID (비우면 전체)", user_ids, default=[])

min_date = df["ActivityDate"].min().date()
max_date = df["ActivityDate"].max().date()
date_range = st.sidebar.date_input(
    "날짜 범위", value=(min_date, max_date), min_value=min_date, max_value=max_date
)

min_wear = st.sidebar.slider(
    "하루 최소 기록 시간(분) - 기기를 거의 안 찬 날 제외",
    min_value=0, max_value=1440, value=0, step=60,
    help="네 활동 시간의 합이 이 값보다 작은 날은 제외합니다.",
)

filtered = df.copy()
if selected_ids:
    filtered = filtered[filtered["Id"].isin(selected_ids)]
if isinstance(date_range, tuple) and len(date_range) == 2:
    start, end = date_range
    filtered = filtered[
        (filtered["ActivityDate"] >= pd.to_datetime(start))
        & (filtered["ActivityDate"] <= pd.to_datetime(end))
    ]
filtered = filtered[filtered["TotalRecordedMinutes"] >= min_wear]

if filtered.empty:
    st.warning("선택한 조건에 해당하는 데이터가 없습니다.")
    st.stop()

# ----------------------------
# 요약 지표
# ----------------------------
st.subheader("📌 요약 지표")
c1, c2, c3, c4, c5 = st.columns(5)
c1.metric("사용자 수", filtered["Id"].nunique())
c2.metric("전체 기록 일수", len(filtered))
c3.metric("평균 걸음 수", f"{filtered['TotalSteps'].mean():,.0f}")
c4.metric("평균 활동 시간(분)", f"{filtered['ActiveMinutes'].mean():,.0f}")
c5.metric("평균 좌식 시간(분)", f"{filtered['SedentaryMinutes'].mean():,.0f}")

# ----------------------------
# 탭 구성
# ----------------------------
tab1, tab2, tab3, tab4, tab5, tab6 = st.tabs(
    ["📋 데이터", "📈 날짜별 추이", "📊 분포·요일", "👥 사용자별 비교", "🧹 데이터 품질", "😴 수면 데이터 연결"]
)

# 탭1: 데이터 확인
with tab1:
    st.write("**데이터 미리보기**")
    st.dataframe(filtered, use_container_width=True)
    st.write("**기초 통계량**")
    st.dataframe(filtered.describe(), use_container_width=True)

# 탭2: 날짜별 추이
with tab2:
    metric = st.selectbox(
        "확인할 항목",
        ["TotalSteps", "VeryActiveMinutes", "FairlyActiveMinutes",
         "LightlyActiveMinutes", "SedentaryMinutes", "ActiveMinutes", "ModerateVigorousMinutes"],
    )
    daily = filtered.groupby("ActivityDate", as_index=False)[metric].mean()
    fig = px.line(daily, x="ActivityDate", y=metric, markers=True, title=f"날짜별 평균 {metric}")
    st.plotly_chart(fig, use_container_width=True)

    st.write("**사용자별 추이 (사용자를 선택하면 더 보기 좋아요)**")
    fig_user = px.line(filtered.sort_values("ActivityDate"), x="ActivityDate", y=metric,
                       color="Id", title=f"사용자별 {metric}")
    st.plotly_chart(fig_user, use_container_width=True)

# 탭3: 분포와 요일
with tab3:
    dist_col = st.selectbox(
        "분포를 볼 항목",
        ["TotalSteps", "VeryActiveMinutes", "FairlyActiveMinutes",
         "LightlyActiveMinutes", "SedentaryMinutes", "ActiveMinutes"],
        key="dist",
    )
    col_a, col_b = st.columns(2)
    with col_a:
        st.plotly_chart(
            px.histogram(filtered, x=dist_col, nbins=30, title=f"{dist_col} 히스토그램"),
            use_container_width=True,
        )
    with col_b:
        wk = filtered.groupby("Weekday", as_index=False)[dist_col].mean()
        st.plotly_chart(
            px.bar(wk, x="Weekday", y=dist_col, category_orders={"Weekday": WEEK_ORDER},
                   title=f"요일별 평균 {dist_col}"),
            use_container_width=True,
        )

    st.write("**변수 간 상관관계 히트맵**")
    corr_cols = ["TotalSteps"] + MINUTE_COLS + ["ActiveMinutes"]
    corr = filtered[corr_cols].corr()
    st.plotly_chart(
        px.imshow(corr, text_auto=".2f", color_continuous_scale="RdBu_r",
                  zmin=-1, zmax=1, aspect="auto"),
        use_container_width=True,
    )

    st.write("**두 변수 관계 (산점도)**")
    cx, cy = st.columns(2)
    x_axis = cx.selectbox("X축", corr_cols, index=0)
    y_axis = cy.selectbox("Y축", corr_cols, index=len(corr_cols) - 1)
    st.plotly_chart(
        px.scatter(filtered, x=x_axis, y=y_axis, color="Id", opacity=0.7,
                   title=f"{x_axis} vs {y_axis}"),
        use_container_width=True,
    )

# 탭4: 사용자별 비교
with tab4:
    st.write("**사용자별 평균 활동 강도(분)**")
    per_user = filtered.groupby("Id", as_index=False)[MINUTE_COLS[:3]].mean()
    melted = per_user.melt(id_vars="Id", var_name="활동 강도", value_name="평균 시간(분)")
    st.plotly_chart(
        px.bar(melted, x="Id", y="평균 시간(분)", color="활동 강도", barmode="stack"),
        use_container_width=True,
    )

    st.write("**사용자별 걸음 수 분포 (박스플롯)**")
    st.plotly_chart(
        px.box(filtered, x="Id", y="TotalSteps", color="Id"),
        use_container_width=True,
    )

    st.write("**사용자별 기록 일수**")
    cnt = filtered.groupby("Id", as_index=False).size().rename(columns={"size": "기록 일수"})
    st.plotly_chart(px.bar(cnt, x="Id", y="기록 일수"), use_container_width=True)

# 탭5: 데이터 품질
with tab5:
    st.info(
        "수면 예측에 쓰기 전에 꼭 확인할 점: SedentaryMinutes(좌식 시간)에는 "
        "잠자는 시간이 포함되어 있을 가능성이 큽니다. 또 기기를 거의 안 찬 날은 "
        "걸음 수가 0에 가깝고 좌식 시간이 1440분에 가깝게 기록돼요."
    )
    st.plotly_chart(
        px.histogram(filtered, x="TotalRecordedMinutes", nbins=30,
                     title="하루 총 기록 시간(분) 분포 (1440 = 24시간)"),
        use_container_width=True,
    )
    low = filtered[filtered["TotalSteps"] < 500]
    st.write(f"**걸음 수 500 미만인 날: {len(low)}일** (기기 미착용 의심)")
    st.dataframe(low, use_container_width=True)

    st.write("**날짜가 비어 있는 구간 (사용자별 누락 일수)**")
    full_days = (filtered["ActivityDate"].max() - filtered["ActivityDate"].min()).days + 1
    miss = filtered.groupby("Id").size().reset_index(name="기록 일수")
    miss["누락 일수"] = full_days - miss["기록 일수"]
    st.dataframe(miss, use_container_width=True)

# 탭6: 수면 데이터 연결
with tab6:
    st.write(
        "`sleepDay_merged.csv`를 올리면 **활동 날짜의 다음 날 수면**과 연결해서 비교합니다. "
        "(예: 4/12 활동 ↔ 4/13 수면)"
    )
    sleep_file = st.file_uploader("sleepDay_merged.csv 업로드", type="csv")

    if sleep_file is None:
        st.info("아직 수면 파일이 없어요. 파일을 올리면 이 탭에 분석이 나타납니다.")
    else:
        try:
            sleep = prepare_sleep(sleep_file)
            merged = filtered.merge(
                sleep[["Id", "SleepDate", "TotalMinutesAsleep", "TotalTimeInBed"]],
                left_on=["Id", "NextDate"], right_on=["Id", "SleepDate"], how="inner",
            )
        except Exception as e:
            st.error(f"수면 데이터를 처리하지 못했습니다: {e}")
            st.stop()

        if merged.empty:
            st.warning("연결된 데이터가 없습니다. 날짜/ID 형식을 확인해 주세요.")
            st.stop()

        st.success(f"연결된 (활동 → 다음 날 수면) 쌍: {len(merged)}개")

        threshold = st.slider(
            "수면 부족 기준 (분) - 이 값보다 적게 자면 수면 부족",
            min_value=240, max_value=540, value=420, step=30,
            help="420분 = 7시간",
        )
        merged["수면부족"] = (merged["TotalMinutesAsleep"] < threshold).map(
            {True: "수면 부족", False: "충분"}
        )

        st.write("**수면 부족 비율**")
        ratio = merged["수면부족"].value_counts().reset_index()
        ratio.columns = ["구분", "일수"]
        st.plotly_chart(px.pie(ratio, names="구분", values="일수"), use_container_width=True)

        feat_cols = ["TotalSteps", "VeryActiveMinutes", "FairlyActiveMinutes",
                     "LightlyActiveMinutes", "SedentaryMinutes", "ActiveMinutes"]
        feat = st.selectbox("전날 활동 변수", feat_cols)

        col_a, col_b = st.columns(2)
        with col_a:
            st.plotly_chart(
                px.box(merged, x="수면부족", y=feat, color="수면부족", points="all",
                       title=f"수면 부족 여부별 전날 {feat}"),
                use_container_width=True,
            )
        with col_b:
            st.plotly_chart(
                px.scatter(merged, x=feat, y="TotalMinutesAsleep", color="수면부족",
                           title=f"전날 {feat} vs 다음 날 수면 시간"),
                use_container_width=True,
            )

        st.write("**전날 활동 변수와 다음 날 수면 시간의 상관계수**")
        corr_sleep = merged[feat_cols + ["TotalMinutesAsleep"]].corr()["TotalMinutesAsleep"].drop(
            "TotalMinutesAsleep"
        ).reset_index()
        corr_sleep.columns = ["활동 변수", "상관계수"]
        st.plotly_chart(
            px.bar(corr_sleep, x="활동 변수", y="상관계수", range_y=[-1, 1]),
            use_container_width=True,
        )
```

상관관계가 낮게 나온 것도 훌륭한 결과예요. 다만 **"상관관계가 없다 ≠ 예측 불가능"**일 수 있어서, 분류 머신러닝으로 이를 더 엄밀하게 검증해볼 수 있습니다.

## 왜 상관계수만으로는 부족할까?

- **상관계수는 직선 관계만 측정**해요. "너무 적게 움직이거나 너무 많이 움직이면 수면이 나빠진다" 같은 U자형 관계는 상관계수가 0에 가깝게 나올 수 있어요.
- **변수 하나씩만 봤어요.** 여러 변수의 조합(예: 걸음 수 높음 + 좌식 시간 높음)은 따로 보면 안 보여요.
- **사람마다 기준이 달라요.** 평소 활동량이 다른 사람들을 한꺼번에 섞어서 봤을 수 있어요.

## 분류 머신러닝으로 할 수 있는 일

1. **여러 변수를 동시에 사용**해서 예측력 확인 (로지스틱 회귀, 랜덤 포레스트)
2. **비선형 관계 탐지**: 트리 기반 모델은 U자형 같은 관계도 잡아내요.
3. **기준선(baseline)과 비교**: "무조건 충분하다고 찍기"보다 나은지 확인
4. **변수 중요도 확인**: 어떤 활동 변수가 그나마 도움이 되는지
5. **"예측이 안 된다"는 결론을 근거 있게 제시**: 이것도 연구 결과예요.

```python
import streamlit as st
import pandas as pd
import numpy as np
import plotly.express as px
from sklearn.model_selection import GroupKFold, cross_val_predict
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier
from sklearn.dummy import DummyClassifier
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import (accuracy_score, f1_score, roc_auc_score,
                             confusion_matrix)

st.set_page_config(page_title="수면 부족 분류 실험", layout="wide")
st.title("🤖 활동 패턴으로 다음 날 수면 부족 분류하기")

ACTIVITY_URL = "https://raw.githubusercontent.com/dg2025825-arch/yeppi/refs/heads/main/data/dailyActivity_merged.csv"
FEATURES = ["TotalSteps", "VeryActiveMinutes", "FairlyActiveMinutes",
            "LightlyActiveMinutes", "SedentaryMinutes"]


@st.cache_data
def load_activity(url):
    df = pd.read_csv(url)
    df["Id"] = df["Id"].astype(str)
    df["ActivityDate"] = pd.to_datetime(df["ActivityDate"], format="%m/%d/%Y")
    df["NextDate"] = df["ActivityDate"] + pd.Timedelta(days=1)
    return df


@st.cache_data
def load_sleep(file):
    sl = pd.read_csv(file)
    sl["Id"] = sl["Id"].astype(str)
    sl["SleepDate"] = pd.to_datetime(sl["SleepDay"], errors="coerce").dt.normalize()
    return sl


act = load_activity(ACTIVITY_URL)
sleep_file = st.file_uploader("sleepDay_merged.csv 업로드", type="csv")
if sleep_file is None:
    st.info("수면 파일을 업로드하면 분석이 시작됩니다.")
    st.stop()

sleep = load_sleep(sleep_file)
data = act.merge(
    sleep[["Id", "SleepDate", "TotalMinutesAsleep"]],
    left_on=["Id", "NextDate"], right_on=["Id", "SleepDate"], how="inner",
)

threshold = st.sidebar.slider("수면 부족 기준(분)", 300, 540, 420, 30)
min_steps = st.sidebar.slider("최소 걸음 수 (기기 미착용 제거)", 0, 3000, 500, 100)

data = data[data["TotalSteps"] >= min_steps].copy()
data["label"] = (data["TotalMinutesAsleep"] < threshold).astype(int)  # 1=수면 부족

st.write(f"사용 데이터: **{len(data)}건**, 사용자 **{data['Id'].nunique()}명**")
ratio = data["label"].mean()
st.write(f"수면 부족 비율: **{ratio:.1%}**")

if data["label"].nunique() < 2 or len(data) < 30:
    st.warning("라벨이 한쪽뿐이거나 데이터가 너무 적습니다. 기준을 조정해 보세요.")
    st.stop()

X = data[FEATURES].values
y = data["label"].values
groups = data["Id"].values
n_splits = min(5, data["Id"].nunique())
cv = GroupKFold(n_splits=n_splits)

models = {
    "기준선(항상 다수 클래스)": DummyClassifier(strategy="most_frequent"),
    "로지스틱 회귀": make_pipeline(
        StandardScaler(), LogisticRegression(class_weight="balanced", max_iter=1000)),
    "랜덤 포레스트": RandomForestClassifier(
        n_estimators=200, max_depth=4, class_weight="balanced", random_state=42),
}

rows, preds = [], {}
for name, model in models.items():
    proba = cross_val_predict(model, X, y, cv=cv, groups=groups, method="predict_proba")[:, 1]
    pred = (proba >= 0.5).astype(int)
    preds[name] = pred
    rows.append({
        "모델": name,
        "정확도": accuracy_score(y, pred),
        "F1": f1_score(y, pred, zero_division=0),
        "AUC": roc_auc_score(y, proba),
    })

result = pd.DataFrame(rows).round(3)
st.subheader("📊 성능 비교 (사용자 단위 교차검증)")
st.dataframe(result, use_container_width=True)
st.caption("AUC 0.5 = 동전 던지기 수준. 기준선보다 뚜렷이 높지 않으면 예측력이 거의 없다는 뜻이에요.")

sel = st.selectbox("혼동행렬을 볼 모델", list(models.keys()), index=2)
cm = confusion_matrix(y, preds[sel])
st.plotly_chart(
    px.imshow(cm, text_auto=True, x=["충분(예측)", "부족(예측)"],
              y=["충분(실제)", "부족(실제)"], color_continuous_scale="Blues",
              title=f"{sel} 혼동행렬"),
    use_container_width=True,
)

st.subheader("🔍 변수 중요도 (랜덤 포레스트)")
rf = RandomForestClassifier(n_estimators=200, max_depth=4,
                            class_weight="balanced", random_state=42).fit(X, y)
imp = pd.DataFrame({"변수": FEATURES, "중요도": rf.feature_importances_})
st.plotly_chart(px.bar(imp.sort_values("중요도"), x="중요도", y="변수", orientation="h"),
                use_container_width=True)
```

아주 좋은 의문이에요. 핵심은 이거예요.

> **상관계수가 측정한 것과, 예측력이 측정하는 것은 다릅니다.**

## 1. 상관계수가 놓칠 수 있는 것

상관계수는 **"변수 하나와 결과가 직선으로 얼마나 같이 움직이는가"**만 봐요. 그래서 아래 경우엔 0에 가깝게 나와도 실제로는 관련이 있을 수 있어요.

**① U자형 관계 (비선형)**
- 걸음 수가 너무 적거나 너무 많으면 수면 부족, 중간이면 충분
- 평균을 내면 직선 경향이 없어서 상관계수 ≈ 0

**② 변수 조합의 효과**
- 걸음 수 높음 **그리고** 좌식 시간 높음일 때만 수면 부족
- 변수를 하나씩 보면 안 보이고, 함께 봐야 보여요.

**③ 사람마다 다른 효과**
- A는 많이 움직이면 잘 자고, B는 많이 움직이면 못 자요.
- 섞으면 서로 상쇄돼서 상관계수 ≈ 0

## 2. "예측력을 확인한다"는 게 무슨 뜻?

방법은 간단해요. **정답을 가린 시험**을 보는 거예요.

1. 데이터를 훈련용 / 시험용으로 나눠요.
2. 훈련용으로 모델이 "이런 활동이면 수면 부족"이라는 규칙을 학습해요.
3. 시험용(모델이 처음 보는 데이터)에서 활동 패턴만 주고 수면 부족인지 맞혀보게 해요.
4. 정답과 비교해서 얼마나 맞혔는지 채점해요.

이때 **"아무 정보 없이 찍는 것(기준선)"보다 잘 맞히는지**가 핵심이에요.

## 3. 결과는 어떻게 해석할까?

| 결과 | 해석 |
|---|---|
| 모델 성능 ≈ 기준선 (AUC ≈ 0.5) | 상관계수와 같은 결론. **직선뿐 아니라 비선형·조합 관계도 없다**고 더 강하게 말할 수 있음 |
| 모델 성능 > 기준선 | 상관계수가 놓친 관계가 있음 |

즉, 관련이 없다고 **이미 믿고 있는 결론을 더 엄밀하게 검증**하는 거예요. 상관계수 하나만으로 "없다"고 하면 "직선만 봤잖아요"라는 반박을 받을 수 있지만, 여러 방법으로 확인하면 결론이 훨씬 탄탄해져요.

## 4. 어떻게 하는가? (앞서 드린 `model_app.py`가 하는 일)

- **로지스틱 회귀**: 여러 변수를 동시에 넣어 "수면 부족일 확률"을 계산하는 직선(선형) 기반 모델. 상관분석의 확장판이에요.
- **랜덤 포레스트**: 질문을 계속 쪼개는 결정 트리 여러 개의 투표. U자형이나 조합 효과도 잡아낼 수 있어요.
- **기준선(Dummy)**: 항상 다수 클래스로 찍는 모델. 비교 대상이에요.
- **GroupKFold**: 사용자 단위로 나눠서 시험. 같은 사람 데이터가 훈련·시험에 섞이지 않게 해요.

실행하면 표에 정확도, F1, AUC가 모델별로 나와요. 거기서 **로지스틱 회귀·랜덤 포레스트가 기준선보다 나은지**만 보면 됩니다.

```python
single_rows = []
for name, model in models.items():
    r = evaluate(model, X[tr_idx], y[tr_idx], X[te_idx], y[te_idx])
    r["모델"] = name
    r["구분"] = "베이스라인" if name in BASELINES else "후보"
    single_rows.append(r)
single = pd.DataFrame(single_rows)[["구분", "모델"] + METRICS].round(3)

st.subheader("📊 ① 고정 분할 결과 (베이스라인 + 후보 한 표)")
st.write(f"학습 {len(tr_idx)}건 / 시험 {len(te_idx)}건 "
         f"(시험 사용자 {len(np.unique(groups[te_idx]))}명, 학습 사용자와 겹치지 않음)")
st.dataframe(single, use_container_width=True, hide_index=True)

# ----------------------------
# ② 반복 분할 (같은 분할을 모든 모델에 적용)
# ----------------------------
rep_rows = []
gss_rep = GroupShuffleSplit(n_splits=n_repeats, test_size=test_size, random_state=SEED)
valid_splits = 0
for i, (tr, te) in enumerate(gss_rep.split(X, y, groups)):
    if len(np.unique(y[tr])) < 2 or len(np.unique(y[te])) < 2:
        continue  # 한 클래스뿐인 분할은 AUC 계산 불가 → 모든 모델에서 동일하게 제외
    valid_splits += 1
    for name, model in models.items():
        r = evaluate(model, X[tr], y[tr], X[te], y[te])
        r["모델"] = name
        r["분할"] = i
        rep_rows.append(r)

if valid_splits == 0:
    st.warning("유효한 반복 분할이 없습니다. 설정을 바꿔보세요.")
    st.stop()

rep = pd.DataFrame(rep_rows)
mean_df = rep.groupby("모델")[METRICS].mean()
std_df = rep.groupby("모델")[METRICS].std()

summary = mean_df.copy().round(3).astype(str) + " ± " + std_df.round(3).astype(str)
summary = summary.loc[BASELINES + CANDIDATES].reset_index()
summary.insert(0, "구분", ["베이스라인"] * len(BASELINES) + ["후보"] * len(CANDIDATES))

st.subheader(f"📊 ② 반복 분할 결과 (유효 분할 {valid_splits}회, 평균 ± 표준편차)")
st.dataframe(summary, use_container_width=True, hide_index=True)
st.caption("같은 분할에서 모든 모델이 시험을 봤어요. ±는 분할이 바뀔 때 점수가 흔들리는 폭이에요.")

# ----------------------------
# 그래프
# ----------------------------
st.subheader("📈 결과 그래프")
long_mean = mean_df.loc[BASELINES + CANDIDATES].reset_index().melt(

st.plotly_chart(
    px.bar(long_mean, x="지표", y="점수", color="모델", barmode="group",

st.plotly_chart(

           title="분할마다 AUC가 얼마나 흔들리는가 (0.5 = 동전 던지기)")
    .add_hline(y=0.5, line_dash="dash"),
    use_container_width=True,

# ----------------------------
# 해석 ① 지표가 우리 데이터에서 의미하는 것
# ----------------------------
st.header("📖 결과 해석")
prev = data["label"].mean()

st.subheader("1. 각 지표는 우리 데이터에서 무슨 뜻인가?")
st.markdown(f"""
(여기서 **양성 = 수면 부족**, 수면 부족 비율은 현재 **{prev:.1%}**)

| 지표 | 우리 데이터에서의 의미 | 왜 중요하게 보는가 |
|---|---|---|
| **정확도** | 전체 (활동일 → 다음 날) 쌍 중 부족/충분을 맞힌 비율 | 직관적이지만 한쪽 클래스가 많으면 속기 쉬워요. 항상 다수 쪽으로 찍어도 약 {max(prev, 1-prev):.0%}가 나와요. |
| **정밀도** | "내일 수면 부족"이라고 경고한 날 중 실제로 부족했던 비율 | 오경보를 얼마나 줄이는지 보여줘요. 낮으면 경고를 믿기 어려워요. |
| **재현율** | 실제 수면 부족이었던 날 중 모델이 잡아낸 비율 | 수면 부족을 놓치지 않는 능력이에요. "항상 부족"이라고 찍으면 100%가 되니 정밀도와 같이 봐야 해요. |
| **F1** | 정밀도와 재현율의 조화평균 | 두 지표 중 하나만 높은 속임수를 막아줘요. 클래스가 쏠린 데이터에서 정확도보다 믿을 만해요. |
| **AUC** | 수면 부족인 날과 충분한 날을 무작위로 하나씩 뽑았을 때, 부족인 날에 더 높은 위험 점수를 줄 확률. 0.5는 동전 던지기, 1은 완벽 | 기준 확률(0.5)에 상관없이 "활동 패턴에 구분 정보가 있는가"를 보여줘요. 우리 질문에 가장 직접적이에요. |
| **균형정확도** | 부족일 때 맞힌 비율(재현율)과 충분일 때 맞힌 비율의 평균 | 클래스 비율이 달라도 공정해요. Dummy는 항상 0.5가 돼요. |
""")

st.subheader("2. 베이스라인보다 나은 모델이 있는가? 차이는 큰가?")

dummy_name, logit_name = BASELINES

for m in CANDIDATES:

st.dataframe(cmp_df, use_container_width=True, hide_index=True)

best_auc = mean_df.loc[CANDIDATES, "AUC"].idxmax()
best_auc_val = mean_df.loc[best_auc, "AUC"]
auc_std = std_df.loc[best_auc, "AUC"]
auc_gain = best_auc_val - 0.5

st.markdown(f"""
**자동 요약**

- 동전 던지기(0.5)보다 **{auc_gain:+.3f}** 높아요 → **{size_label(auc_gain, auc_std)}**.
- 이 판단은 "차이가 분할마다 흔들리는 폭(표준편차)보다 큰가?"와 대략적인 크기 기준(0.05, 0.10)으로 한 거예요. 통계적 검정이 아닌 **참고용 기준**이에요.

**읽는 법**
- 평균 차이가 표준편차보다 작으면, 분할을 바꿨을 때 순위가 뒤바뀔 수 있어서 "더 낫다"고 말하기 어려워요.

""")

if n_unique_winners == 1:
```

```python
import os
import streamlit as st
import pandas as pd
import numpy as np
import plotly.express as px
from sklearn.base import clone
from sklearn.model_selection import GroupShuffleSplit
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.neighbors import KNeighborsClassifier
from sklearn.dummy import DummyClassifier
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import (accuracy_score, precision_score, recall_score,
                             f1_score, roc_auc_score, balanced_accuracy_score)

# ----------------------------
# 기본 설정
# ----------------------------
st.set_page_config(page_title="모델 비교", layout="wide")
st.title("🧪 베이스라인 vs 후보 모델 비교")
st.caption("질문: 하루의 활동 패턴만으로 다음 날 수면 부족 여부를 예측할 수 있는가?")

SEED = 42

# 경로: 이 파일(code/) 기준으로 상위 폴더의 data/ 를 가리킴
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
DATA_DIR = os.path.join(BASE_DIR, "..", "data")
ACTIVITY_PATH = os.path.join(DATA_DIR, "dailyActivity_merged.csv")
SLEEP_PATH = os.path.join(DATA_DIR, "sleepDay_merged.csv")

FEATURES = ["TotalSteps", "VeryActiveMinutes", "FairlyActiveMinutes",
            "LightlyActiveMinutes", "SedentaryMinutes"]
METRICS = ["정확도", "정밀도", "재현율", "F1", "AUC", "균형정확도"]

BASELINES = ["Dummy(다수 클래스)", "로지스틱 회귀"]
CANDIDATES = ["랜덤 포레스트", "그래디언트 부스팅", "KNN"]


def get_models():
    return {
        "Dummy(다수 클래스)": DummyClassifier(strategy="most_frequent"),
        "로지스틱 회귀": make_pipeline(
            StandardScaler(),
            LogisticRegression(class_weight="balanced", max_iter=1000, random_state=SEED)),
        "랜덤 포레스트": RandomForestClassifier(
            n_estimators=200, max_depth=4, class_weight="balanced", random_state=SEED),
        "그래디언트 부스팅": GradientBoostingClassifier(
            n_estimators=100, max_depth=2, random_state=SEED),
        "KNN": make_pipeline(StandardScaler(), KNeighborsClassifier(n_neighbors=15)),
    }


# ----------------------------
# 데이터 불러오기
# ----------------------------
@st.cache_data
def load_activity(path):
    df = pd.read_csv(path)
    df["Id"] = df["Id"].astype(str)
    df["ActivityDate"] = pd.to_datetime(df["ActivityDate"], format="%m/%d/%Y")
    df["NextDate"] = df["ActivityDate"] + pd.Timedelta(days=1)
    return df


@st.cache_data
def load_sleep(path_or_file):
    sl = pd.read_csv(path_or_file)
    sl["Id"] = sl["Id"].astype(str)
    sl["SleepDate"] = pd.to_datetime(sl["SleepDay"], errors="coerce").dt.normalize()
    return sl


def evaluate(model, X_tr, y_tr, X_te, y_te):
    m = clone(model).fit(X_tr, y_tr)
    proba = m.predict_proba(X_te)[:, 1]
    pred = (proba >= 0.5).astype(int)
    return {
        "정확도": accuracy_score(y_te, pred),
        "정밀도": precision_score(y_te, pred, zero_division=0),
        "재현율": recall_score(y_te, pred, zero_division=0),
        "F1": f1_score(y_te, pred, zero_division=0),
        "AUC": roc_auc_score(y_te, proba),
        "균형정확도": balanced_accuracy_score(y_te, pred),
    }


def size_label(diff, std):
    """차이의 크기를 말로 표현 (대략적인 기준, 절대적 규칙 아님)"""
    if diff <= 0:
        return "베이스라인보다 낫지 않음"
    if diff <= std:
        return "차이가 작음 (분할마다 흔들리는 폭 안쪽이라 우연일 수 있음)"
    if diff < 0.05:
        return "작은 편"
    if diff < 0.10:
        return "중간 정도"
    return "큰 편"


# ----------------------------
# 데이터 준비
# ----------------------------
if not os.path.exists(ACTIVITY_PATH):
    st.error(f"활동 데이터를 찾을 수 없습니다: {os.path.normpath(ACTIVITY_PATH)}")
    st.stop()
act = load_activity(ACTIVITY_PATH)

if os.path.exists(SLEEP_PATH):
    sleep = load_sleep(SLEEP_PATH)
    st.success("data/sleepDay_merged.csv 를 불러왔어요.")
else:
    st.warning("data/sleepDay_merged.csv 가 없어요. 파일을 직접 업로드해 주세요.")
    sleep_file = st.file_uploader("sleepDay_merged.csv 업로드", type="csv")
    if sleep_file is None:
        st.stop()
    sleep = load_sleep(sleep_file)

data = act.merge(
    sleep[["Id", "SleepDate", "TotalMinutesAsleep"]],
    left_on=["Id", "NextDate"], right_on=["Id", "SleepDate"], how="inner",
)

st.sidebar.header("⚙️ 설정")
threshold = st.sidebar.slider("수면 부족 기준(분)", 300, 540, 420, 30)
min_steps = st.sidebar.slider("최소 걸음 수 (기기 미착용 제거)", 0, 3000, 500, 100)
test_size = st.sidebar.slider("시험 데이터 비율(사용자 기준)", 0.2, 0.5, 0.3, 0.05)
n_repeats = st.sidebar.slider("반복 분할 횟수", 5, 50, 20, 5)
st.sidebar.write(f"난수 고정값: **{SEED}**")

data = data[data["TotalSteps"] >= min_steps].copy()
data["label"] = (data["TotalMinutesAsleep"] < threshold).astype(int)  # 1 = 수면 부족

st.write(f"사용 데이터: **{len(data)}건**, 사용자 **{data['Id'].nunique()}명**, "
         f"수면 부족 비율 **{data['label'].mean():.1%}**")

if data["label"].nunique() < 2 or len(data) < 50 or data["Id"].nunique() < 6:
    st.warning("라벨이 한쪽뿐이거나 데이터가 너무 적습니다. 기준을 조정해 보세요.")
    st.stop()

X = data[FEATURES].values
y = data["label"].values
groups = data["Id"].values
models = get_models()

# ----------------------------
# ① 고정 분할 1회 (모든 모델 동일 분할)
# ----------------------------
gss = GroupShuffleSplit(n_splits=1, test_size=test_size, random_state=SEED)
tr_idx, te_idx = next(gss.split(X, y, groups))

if len(np.unique(y[tr_idx])) < 2 or len(np.unique(y[te_idx])) < 2:
    st.warning("고정 분할의 학습 또는 시험 데이터에 한 클래스만 있어요. 기준이나 시험 비율을 바꿔보세요.")
    st.stop()

single_rows = []
for name, model in models.items():
    r = evaluate(model, X[tr_idx], y[tr_idx], X[te_idx], y[te_idx])
    r["모델"] = name
    r["구분"] = "베이스라인" if name in BASELINES else "후보"
    single_rows.append(r)
single = pd.DataFrame(single_rows)[["구분", "모델"] + METRICS].round(3)

st.subheader("📊 ① 고정 분할 결과 (베이스라인 + 후보 한 표)")
st.write(f"학습 {len(tr_idx)}건 / 시험 {len(te_idx)}건 "
         f"(시험 사용자 {len(np.unique(groups[te_idx]))}명, 학습 사용자와 겹치지 않음)")
st.dataframe(single, use_container_width=True, hide_index=True)

# ----------------------------
# ② 반복 분할 (같은 분할을 모든 모델에 적용)
# ----------------------------
rep_rows = []
gss_rep = GroupShuffleSplit(n_splits=n_repeats, test_size=test_size, random_state=SEED)
valid_splits = 0
for i, (tr, te) in enumerate(gss_rep.split(X, y, groups)):
    if len(np.unique(y[tr])) < 2 or len(np.unique(y[te])) < 2:
        continue  # 한 클래스뿐인 분할은 AUC 계산 불가 → 모든 모델에서 동일하게 제외
    valid_splits += 1
    for name, model in models.items():
        r = evaluate(model, X[tr], y[tr], X[te], y[te])
        r["모델"] = name
        r["분할"] = i
        rep_rows.append(r)

if valid_splits == 0:
    st.warning("유효한 반복 분할이 없습니다. 설정을 바꿔보세요.")
    st.stop()

rep = pd.DataFrame(rep_rows)
mean_df = rep.groupby("모델")[METRICS].mean()
std_df = rep.groupby("모델")[METRICS].std()

summary = mean_df.copy().round(3).astype(str) + " ± " + std_df.round(3).astype(str)
summary = summary.loc[BASELINES + CANDIDATES].reset_index()
summary.insert(0, "구분", ["베이스라인"] * len(BASELINES) + ["후보"] * len(CANDIDATES))

st.subheader(f"📊 ② 반복 분할 결과 (유효 분할 {valid_splits}회, 평균 ± 표준편차)")
st.dataframe(summary, use_container_width=True, hide_index=True)
st.caption("같은 분할에서 모든 모델이 시험을 봤어요. ±는 분할이 바뀔 때 점수가 흔들리는 폭이에요.")

# ----------------------------
# 그래프
# ----------------------------
st.subheader("📈 결과 그래프")
long_mean = mean_df.loc[BASELINES + CANDIDATES].reset_index().melt(
    id_vars="모델", var_name="지표", value_name="점수")
st.plotly_chart(
    px.bar(long_mean, x="지표", y="점수", color="모델", barmode="group",
           category_orders={"모델": BASELINES + CANDIDATES},
           title="지표별 평균 점수 (반복 분할)", range_y=[0, 1]),
    use_container_width=True,
)

st.plotly_chart(
    px.box(rep, x="모델", y="AUC", color="모델", points="all",
           category_orders={"모델": BASELINES + CANDIDATES},
           title="분할마다 AUC가 얼마나 흔들리는가 (0.5 = 동전 던지기)")
    .add_hline(y=0.5, line_dash="dash"),
    use_container_width=True,

# ----------------------------
# 해석 ① 지표가 우리 데이터에서 의미하는 것
# ----------------------------
st.header("📖 결과 해석")
prev = data["label"].mean()

st.subheader("1. 각 지표는 우리 데이터에서 무슨 뜻인가?")
st.markdown(f"""
(여기서 **양성 = 수면 부족**, 수면 부족 비율은 현재 **{prev:.1%}**)

| 지표 | 우리 데이터에서의 의미 | 왜 중요하게 보는가 |
|---|---|---|
| **정확도** | 전체 (활동일 → 다음 날) 쌍 중 부족/충분을 맞힌 비율 | 직관적이지만 한쪽 클래스가 많으면 속기 쉬워요. 항상 다수 쪽으로 찍어도 약 {max(prev, 1-prev):.0%}가 나와요. |
| **정밀도** | "내일 수면 부족"이라고 경고한 날 중 실제로 부족했던 비율 | 오경보를 얼마나 줄이는지 보여줘요. 낮으면 경고를 믿기 어려워요. |
| **재현율** | 실제 수면 부족이었던 날 중 모델이 잡아낸 비율 | 수면 부족을 놓치지 않는 능력이에요. "항상 부족"이라고 찍으면 100%가 되니 정밀도와 같이 봐야 해요. |

| **AUC** | 수면 부족인 날과 충분한 날을 무작위로 하나씩 뽑았을 때, 부족인 날에 더 높은 위험 점수를 줄 확률. 0.5는 동전 던지기, 1은 완벽 | 기준 확률(0.5)에 상관없이 "활동 패턴에 구분 정보가 있는가"를 보여줘요. 우리 질문에 가장 직접적이에요. |
| **균형정확도** | 부족일 때 맞힌 비율(재현율)과 충분일 때 맞힌 비율의 평균 | 클래스 비율이 달라도 공정해요. Dummy는 항상 0.5가 돼요. |
""")

dummy_name, logit_name = BASELINES
rows = []
for m in CANDIDATES:

- 동전 던지기(0.5)보다 **{auc_gain:+.3f}** 높아요 → **{size_label(auc_gain, auc_std)}**.
- 이 판단은 "차이가 분할마다 흔들리는 폭(표준편차)보다 큰가?"와 대략적인 크기 기준(0.05, 0.10)으로 한 거예요. 통계적 검정이 아닌 **참고용 기준**이에요.

**읽는 법**
- 평균 차이가 표준편차보다 작으면, 분할을 바꿨을 때 순위가 뒤바뀔 수 있어서 "더 낫다"고 말하기 어려워요.

""")

for metric in METRICS:

    verdict = f"모든 지표에서 1위가 **{rank_df['1위 모델'].iloc[0]}**로 같아요. 지표에 따라 결론이 달라지지 않았어요."
else:

""")

st.info(
    "한계: 사용자 수와 데이터 수가 적고, 수면 부족 기준(분)이나 기기 미착용 제거 기준을 바꾸면 결과가 달라질 수 있어요. "
    "사이드바의 기준을 바꿔서 결론이 일관적인지 꼭 확인해 보세요. "
    "그래디언트 부스팅과 KNN은 클래스 비율 보정을 쓰지 않아 재현율이 낮게 나올 수 있어요."
)
```

```python
import os
import numpy as np
import pandas as pd
import streamlit as st
from sklearn.base import clone
from sklearn.model_selection import GroupShuffleSplit
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.neighbors import KNeighborsClassifier
from sklearn.dummy import DummyClassifier
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import (accuracy_score, precision_score, recall_score,
                             f1_score, roc_auc_score, balanced_accuracy_score)

SEED = 42

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
DATA_DIR = os.path.join(BASE_DIR, "..", "data")
ACTIVITY_PATH = os.path.join(DATA_DIR, "dailyActivity_merged.csv")
SLEEP_PATH = os.path.join(DATA_DIR, "sleepDay_merged.csv")

FEATURES = ["TotalSteps", "VeryActiveMinutes", "FairlyActiveMinutes",
            "LightlyActiveMinutes", "SedentaryMinutes"]
METRICS = ["정확도", "정밀도", "재현율", "F1", "AUC", "균형정확도"]
BASELINES = ["Dummy(다수 클래스)", "로지스틱 회귀"]
CANDIDATES = ["랜덤 포레스트", "그래디언트 부스팅", "KNN"]
ALL_MODELS = BASELINES + CANDIDATES

SETTING_KEYS = ["threshold", "min_steps", "test_size", "n_repeats"]


def get_models():
    return {
        "Dummy(다수 클래스)": DummyClassifier(strategy="most_frequent"),
        "로지스틱 회귀": make_pipeline(
            StandardScaler(),
            LogisticRegression(class_weight="balanced", max_iter=1000, random_state=SEED)),
        "랜덤 포레스트": RandomForestClassifier(
            n_estimators=200, max_depth=4, class_weight="balanced", random_state=SEED),
        "그래디언트 부스팅": GradientBoostingClassifier(
            n_estimators=100, max_depth=2, random_state=SEED),
        "KNN": make_pipeline(StandardScaler(), KNeighborsClassifier(n_neighbors=15)),
    }


@st.cache_data
def load_activity(path):
    df = pd.read_csv(path)
    df["Id"] = df["Id"].astype(str)
    df["ActivityDate"] = pd.to_datetime(df["ActivityDate"], format="%m/%d/%Y")
    df["NextDate"] = df["ActivityDate"] + pd.Timedelta(days=1)
    return df


def prepare_sleep(sl):
    sl = sl.copy()
    sl["Id"] = sl["Id"].astype(str)
    sl["SleepDate"] = pd.to_datetime(sl["SleepDay"], errors="coerce").dt.normalize()
    return sl


def evaluate(model, X_tr, y_tr, X_te, y_te):
    m = clone(model).fit(X_tr, y_tr)
    proba = m.predict_proba(X_te)[:, 1]
    pred = (proba >= 0.5).astype(int)
    return {
        "정확도": accuracy_score(y_te, pred),
        "정밀도": precision_score(y_te, pred, zero_division=0),
        "재현율": recall_score(y_te, pred, zero_division=0),
        "F1": f1_score(y_te, pred, zero_division=0),
        "AUC": roc_auc_score(y_te, proba),
        "균형정확도": balanced_accuracy_score(y_te, pred),
    }


def size_label(diff, std):
    """차이의 크기를 말로 표현 (대략적인 기준, 절대적 규칙 아님)"""
    if diff <= 0:
        return "베이스라인보다 낫지 않음"
    if diff <= std:
        return "차이가 작음 (분할마다 흔들리는 폭 안쪽이라 우연일 수 있음)"
    if diff < 0.05:
        return "작은 편"
    if diff < 0.10:
        return "중간 정도"
    return "큰 편"


@st.cache_data(show_spinner="모델을 학습하고 평가하는 중...")
def run_experiment(X, y, groups, test_size, n_repeats):
    """모든 모델이 같은 분할로 시험을 본다."""
    models = get_models()

    # 고정 분할 1회
    gss = GroupShuffleSplit(n_splits=1, test_size=test_size, random_state=SEED)
    tr_idx, te_idx = next(gss.split(X, y, groups))
    if len(np.unique(y[tr_idx])) < 2 or len(np.unique(y[te_idx])) < 2:
        return None

    single_rows = []
    for name, model in models.items():
        r = evaluate(model, X[tr_idx], y[tr_idx], X[te_idx], y[te_idx])
        r["모델"] = name
        r["구분"] = "베이스라인" if name in BASELINES else "후보"
        single_rows.append(r)
    single = pd.DataFrame(single_rows)[["구분", "모델"] + METRICS]

    # 반복 분할
    rep_rows = []
    valid = 0
    gss_rep = GroupShuffleSplit(n_splits=n_repeats, test_size=test_size, random_state=SEED)
    for i, (tr, te) in enumerate(gss_rep.split(X, y, groups)):
        if len(np.unique(y[tr])) < 2 or len(np.unique(y[te])) < 2:
            continue
        valid += 1
        for name, model in models.items():
            r = evaluate(model, X[tr], y[tr], X[te], y[te])
            r["모델"] = name
            r["분할"] = i
            rep_rows.append(r)
    if valid == 0:
        return None

    return {"single": single, "rep": pd.DataFrame(rep_rows),
            "valid": valid, "tr_idx": tr_idx, "te_idx": te_idx}


def setup():
    """모든 페이지 공통: 설정 사이드바, 데이터 준비, 실험 실행. 결과 dict 반환."""
    # 페이지를 옮겨도 사이드바 설정이 유지되도록 하는 처리
    for k in SETTING_KEYS:
        if k in st.session_state:
            st.session_state[k] = st.session_state[k]

    if not os.path.exists(ACTIVITY_PATH):
        st.error(f"활동 데이터를 찾을 수 없습니다: {os.path.normpath(ACTIVITY_PATH)}")
        st.stop()
    act = load_activity(ACTIVITY_PATH)

    if "sleep_df" not in st.session_state:
        if os.path.exists(SLEEP_PATH):
            st.session_state["sleep_df"] = pd.read_csv(SLEEP_PATH)
        else:
            st.warning("data/sleepDay_merged.csv 가 없어요. 파일을 직접 업로드해 주세요.")
            up = st.file_uploader("sleepDay_merged.csv 업로드", type="csv")
            if up is None:
                st.stop()
            st.session_state["sleep_df"] = pd.read_csv(up)
            st.rerun()
    sleep = prepare_sleep(st.session_state["sleep_df"])

    data = act.merge(
        sleep[["Id", "SleepDate", "TotalMinutesAsleep"]],
        left_on=["Id", "NextDate"], right_on=["Id", "SleepDate"], how="inner",
    )

    st.sidebar.header("⚙️ 설정")
    threshold = st.sidebar.slider("수면 부족 기준(분)", 300, 540, 420, 30, key="threshold")
    min_steps = st.sidebar.slider("최소 걸음 수 (기기 미착용 제거)", 0, 3000, 500, 100, key="min_steps")
    test_size = st.sidebar.slider("시험 데이터 비율(사용자 기준)", 0.2, 0.5, 0.3, 0.05, key="test_size")
    n_repeats = st.sidebar.slider("반복 분할 횟수", 5, 50, 20, 5, key="n_repeats")
    st.sidebar.write(f"난수 고정값: **{SEED}**")

    data = data[data["TotalSteps"] >= min_steps].copy()
    data["label"] = (data["TotalMinutesAsleep"] < threshold).astype(int)  # 1 = 수면 부족

    if data["label"].nunique() < 2 or len(data) < 50 or data["Id"].nunique() < 6:
        st.warning("라벨이 한쪽뿐이거나 데이터가 너무 적습니다. 사이드바 기준을 조정해 보세요.")
        st.stop()

    X = data[FEATURES].values
    y = data["label"].values
    groups = data["Id"].values

    res = run_experiment(X, y, groups, test_size, n_repeats)
    if res is None:
        st.warning("학습 또는 시험 데이터에 한 클래스만 있는 분할이에요. 기준이나 시험 비율을 바꿔보세요.")
        st.stop()

    rep = res["rep"]
    return {
        "data": data, "X": X, "y": y, "groups": groups,
        "prev": data["label"].mean(), "threshold": threshold,
        "single": res["single"], "rep": rep, "valid": res["valid"],
        "tr_idx": res["tr_idx"], "te_idx": res["te_idx"],
        "mean_df": rep.groupby("모델")[METRICS].mean(),
        "std_df": rep.groupby("모델")[METRICS].std(),
    }
```

## 한 줄 결론

> **모델이 활동 패턴만 보고는 "수면 부족인 날"과 "충분한 날"을 동전 던지기 수준으로밖에 구분하지 못했다**는 뜻이에요.

## 우리 데이터로 풀어서 읽으면

수면 부족이었던 날의 전날 활동 기록 하나, 충분히 잤던 날의 전날 활동 기록 하나를 뽑아 모델에게 보여준다고 해요. 모델이 "이 중 어느 쪽이 다음 날 수면 부족일 것 같아?"를 점수로 매기면:

- AUC 1.0 → 항상 부족인 날의 점수가 더 높음 (완벽)
- AUC 0.5 → 절반은 맞고 절반은 틀림 (그냥 찍는 것과 같음)
- AUC 0.5 근처 → **활동 패턴(걸음 수, 활동 시간)에 수면 부족을 알려주는 정보가 거의 없음**

그래서 모든 모델이 0.5 근처라면, 로지스틱 회귀(직선 관계)든 랜덤 포레스트(비선형·조합 관계)든 **어떤 방식으로 봐도 단서를 못 찾았다**는 의미예요. 앞서 상관계수가 낮게 나온 결과와 같은 방향의 결론이고, 더 엄밀하게 확인한 셈이에요.

## 연구 결론으로 쓴다면

> "이 데이터에서는 하루의 활동 패턴만으로 다음 날 수면 부족 여부를 예측하기 어려웠다. 선형·비선형 모델 모두 AUC가 0.5 부근이었다."

이것은 실패가 아니라 **"예측 불가능하다"는 근거 있는 결과**예요.

## 다만 조심할 점 (과장 금지)

"활동과 수면은 전혀 관계없다"까지 말하는 건 지나쳐요. 가능한 다른 이유들이 있어요.

1. **데이터가 적음**: 수면 기록이 있는 사용자가 약 24명, 400여 건이라 약한 관계는 잡히지 않을 수 있어요.
2. **변수가 제한적**: 하루 합계(걸음 수, 활동 분)만 썼어요. 활동 시각(밤늦은 운동 등)은 없어요.
3. **좌식 시간에 수면이 섞임**: `SedentaryMinutes`에 잠자는 시간이 포함돼 있을 수 있어 신호가 흐려질 수 있어요.
4. **수면 부족 기준**: 7시간 기준이 적절한지에 따라 결과가 달라질 수 있어요.
5. **우연의 범위**: AUC가 0.5에서 0.03~0.05 벗어난 정도는 분할마다 흔들리는 폭 안쪽일 수 있어요. 표준편차를 같이 확인하세요.

그래서 정확한 표현은 **"관계가 없다"가 아니라 "이 데이터와 이 변수로는 예측에 쓸 만한 신호를 찾지 못했다"**예요.

## 설계 요약

| 항목 | 내용 |
|---|---|
| 모델 | RandomForestClassifier (난수 42 고정) |
| 입력 변수 | 걸음 수, 활동 강도별 시간 4개 |
| 정답(라벨) | 다음 날 수면 시간이 기준 미만이면 1(수면 부족) |
| 분할 | 사용자 단위(같은 사람이 학습·시험에 섞이지 않음) |
| 튜닝 | 하이퍼파라미터를 사용자 단위 교차검증(GroupKFold)으로 선택 |
| 평가 | 시험 데이터에서 AUC, 균형정확도, F1 등 + 기준선(Dummy) 비교 |
| 해석 | 변수 중요도, 순열 중요도, 혼동행렬 |

튜닝은 학습 데이터 안에서만 하고 시험 데이터는 마지막에 한 번만 써요. 시험 데이터로 튜닝하면 성능이 부풀려지기 때문이에요.

## code/rf_model.py

```python
import os
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.dummy import DummyClassifier
from sklearn.model_selection import GroupShuffleSplit, GroupKFold, GridSearchCV
from sklearn.inspection import permutation_importance
from sklearn.metrics import (accuracy_score, precision_score, recall_score,
                             f1_score, roc_auc_score, balanced_accuracy_score,
                             confusion_matrix)

SEED = 42

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
DATA_DIR = os.path.join(BASE_DIR, "..", "data")
ACTIVITY_PATH = os.path.join(DATA_DIR, "dailyActivity_merged.csv")
SLEEP_PATH = os.path.join(DATA_DIR, "sleepDay_merged.csv")

FEATURES = ["TotalSteps", "VeryActiveMinutes", "FairlyActiveMinutes",
            "LightlyActiveMinutes", "SedentaryMinutes"]


def load_dataset(threshold=420, min_steps=500):
    """활동(전날) + 수면(다음 날)을 합치고 수면 부족 라벨(1)을 만든다."""
    act = pd.read_csv(ACTIVITY_PATH)
    act["Id"] = act["Id"].astype(str)
    act["ActivityDate"] = pd.to_datetime(act["ActivityDate"], format="%m/%d/%Y")
    act["NextDate"] = act["ActivityDate"] + pd.Timedelta(days=1)

    sl = pd.read_csv(SLEEP_PATH)
    sl["Id"] = sl["Id"].astype(str)
    sl["SleepDate"] = pd.to_datetime(sl["SleepDay"], errors="coerce").dt.normalize()

    data = act.merge(
        sl[["Id", "SleepDate", "TotalMinutesAsleep"]],
        left_on=["Id", "NextDate"], right_on=["Id", "SleepDate"], how="inner",
    )
    data = data[data["TotalSteps"] >= min_steps].copy()
    data["label"] = (data["TotalMinutesAsleep"] < threshold).astype(int)
    return data.reset_index(drop=True)


def _metrics(y_true, proba, thr=0.5):
    pred = (proba >= thr).astype(int)
    return {
        "정확도": accuracy_score(y_true, pred),
        "정밀도": precision_score(y_true, pred, zero_division=0),
        "재현율": recall_score(y_true, pred, zero_division=0),
        "F1": f1_score(y_true, pred, zero_division=0),
        "AUC": roc_auc_score(y_true, proba),
        "균형정확도": balanced_accuracy_score(y_true, pred),
    }


def train_random_forest(data, test_size=0.3, tune=True, cv_splits=4):
    """사용자 단위로 분할 → 학습 데이터 안에서 튜닝 → 시험 데이터로 최종 평가."""
    X = data[FEATURES].values
    y = data["label"].values
    groups = data["Id"].values

    gss = GroupShuffleSplit(n_splits=1, test_size=test_size, random_state=SEED)
    tr, te = next(gss.split(X, y, groups))
    if len(np.unique(y[tr])) < 2 or len(np.unique(y[te])) < 2:
        raise ValueError("학습 또는 시험 데이터에 한 클래스만 있어요. 기준/시험 비율을 바꿔보세요.")

    base_rf = RandomForestClassifier(class_weight="balanced", random_state=SEED, n_jobs=-1)

    best_params, cv_table = {}, None
    if tune:
        grid = {
            "n_estimators": [100, 300],
            "max_depth": [2, 3, 4, 6],
            "min_samples_leaf": [5, 10, 20],
        }
        n_groups = len(np.unique(groups[tr]))
        gs = GridSearchCV(
            base_rf, grid, scoring="roc_auc",
            cv=GroupKFold(n_splits=min(cv_splits, n_groups)), n_jobs=-1,
        )
        gs.fit(X[tr], y[tr], groups=groups[tr])
        model = gs.best_estimator_
        best_params = gs.best_params_
        cv_table = (pd.DataFrame(gs.cv_results_)
                    [["param_n_estimators", "param_max_depth", "param_min_samples_leaf",
                      "mean_test_score", "std_test_score"]]
                    .rename(columns={"param_n_estimators": "트리 수",
                                     "param_max_depth": "최대 깊이",
                                     "param_min_samples_leaf": "리프 최소 표본",
                                     "mean_test_score": "교차검증 AUC 평균",
                                     "std_test_score": "표준편차"})
                    .sort_values("교차검증 AUC 평균", ascending=False)
                    .reset_index(drop=True))
    else:
        model = RandomForestClassifier(
            n_estimators=200, max_depth=4, min_samples_leaf=10,
            class_weight="balanced", random_state=SEED, n_jobs=-1,
        ).fit(X[tr], y[tr])
        best_params = {"n_estimators": 200, "max_depth": 4, "min_samples_leaf": 10}

    dummy = DummyClassifier(strategy="most_frequent").fit(X[tr], y[tr])

    rf_proba = model.predict_proba(X[te])[:, 1]
    dm_proba = dummy.predict_proba(X[te])[:, 1]
    scores = pd.DataFrame({
        "Dummy(다수 클래스)": _metrics(y[te], dm_proba),
        "랜덤 포레스트": _metrics(y[te], rf_proba),
    }).T

    perm = permutation_importance(
        model, X[te], y[te], scoring="roc_auc",
        n_repeats=30, random_state=SEED, n_jobs=-1,
    )
    importance = pd.DataFrame({
        "변수": FEATURES,
        "트리 중요도": model.feature_importances_,
        "순열 중요도(AUC 감소량)": perm.importances_mean,
        "순열 중요도 표준편차": perm.importances_std,
    }).sort_values("순열 중요도(AUC 감소량)", ascending=False).reset_index(drop=True)

    return {
        "model": model, "best_params": best_params, "cv_table": cv_table,
        "scores": scores, "importance": importance,
        "cm": confusion_matrix(y[te], (rf_proba >= 0.5).astype(int)),
        "y_te": y[te], "proba_te": rf_proba,
        "n_train": len(tr), "n_test": len(te),
        "train_users": len(np.unique(groups[tr])), "test_users": len(np.unique(groups[te])),
    }


def predict_one(model, steps, very, fairly, lightly, sedentary):
    """하루 활동 입력 → 다음 날 수면 부족 확률."""
    x = np.array([[steps, very, fairly, lightly, sedentary]])
    return float(model.predict_proba(x)[0, 1])


if __name__ == "__main__":
    d = load_dataset()
    print(f"데이터 {len(d)}건, 사용자 {d['Id'].nunique()}명, 수면 부족 비율 {d['label'].mean():.1%}")
    r = train_random_forest(d)
    print("선택된 하이퍼파라미터:", r["best_params"])
    print(r["scores"].round(3))
    print(r["importance"].round(3))
```

## 코드에서 짚어둘 점

- **누수 방지**: 튜닝은 학습 데이터 안의 사용자 단위 교차검증으로만 하고, 시험 데이터는 마지막 평가에만 써요.
- **`class_weight="balanced"`**: 수면 부족/충분 비율이 쏠려도 한쪽만 찍지 않게 보정해요.
- **`min_samples_leaf`, `max_depth` 제한**: 표본이 작아서 트리를 깊게 키우면 과적합돼요.
- **순열 중요도**: 시험 데이터에서 변수를 섞었을 때 AUC가 얼마나 떨어지는지 보는 방식이라, 트리 중요도보다 믿을 만해요.
- 위 입력 예측 화면은 도구 사용 예시일 뿐, AUC가 0.5 근처인 결과에서는 실제 예측 도구로 보면 안 돼요.

```python
    sl["SleepDate"] = pd.to_datetime(sl["SleepDay"], errors="coerce").dt.normalize()

    data = act.merge(
        sl[["Id", "SleepDate", "TotalMinutesAsleep"]],
        left_on=["Id", "NextDate"], right_on=["Id", "SleepDate"], how="inner",
    )
    data = data[data["TotalSteps"] >= min_steps].copy()
    data["label"] = (data["TotalMinutesAsleep"] < threshold).astype(int)
    return data.reset_index(drop=True)


def calc_metrics(y_true, proba, thr=0.5):
    pred = (proba >= thr).astype(int)
    return {
        "정확도": accuracy_score(y_true, pred),
        "정밀도": precision_score(y_true, pred, zero_division=0),
        "재현율": recall_score(y_true, pred, zero_division=0),
        "F1": f1_score(y_true, pred, zero_division=0),
        "AUC": roc_auc_score(y_true, proba),
        "균형정확도": balanced_accuracy_score(y_true, pred),
    }


@st.cache_data(show_spinner="랜덤 포레스트를 학습하는 중입니다...")
def train(threshold, min_steps, test_size, tune):
    data = load_dataset(threshold, min_steps)
    if data["label"].nunique() < 2 or len(data) < 50 or data["Id"].nunique() < 6:
        return None

    X = data[FEATURES].values
    y = data["label"].values
    groups = data["Id"].values

    # 사용자 단위 분할 (같은 사람이 학습/시험에 섞이지 않음)
    gss = GroupShuffleSplit(n_splits=1, test_size=test_size, random_state=SEED)
    tr, te = next(gss.split(X, y, groups))
    if len(np.unique(y[tr])) < 2 or len(np.unique(y[te])) < 2:
        return None

    base_rf = RandomForestClassifier(class_weight="balanced", random_state=SEED, n_jobs=-1)
    cv_table = None
    if tune:
        grid = {"n_estimators": [100, 300],
                "max_depth": [2, 3, 4, 6],
                "min_samples_leaf": [5, 10, 20]}
        n_groups = len(np.unique(groups[tr]))
        gs = GridSearchCV(base_rf, grid, scoring="roc_auc",
                          cv=GroupKFold(n_splits=min(4, n_groups)), n_jobs=-1)
        gs.fit(X[tr], y[tr], groups=groups[tr])
        model, best_params = gs.best_estimator_, gs.best_params_
        cv_table = (pd.DataFrame(gs.cv_results_)
                    [["param_n_estimators", "param_max_depth", "param_min_samples_leaf",
                      "mean_test_score", "std_test_score"]]
                    .rename(columns={"param_n_estimators": "트리 수",
                                     "param_max_depth": "최대 깊이",
                                     "param_min_samples_leaf": "리프 최소 표본",
                                     "mean_test_score": "교차검증 AUC 평균",
                                     "std_test_score": "표준편차"})
                    .sort_values("교차검증 AUC 평균", ascending=False)
                    .reset_index(drop=True))
    else:
        best_params = {"n_estimators": 200, "max_depth": 4, "min_samples_leaf": 10}
        model = RandomForestClassifier(class_weight="balanced", random_state=SEED,
                                       n_jobs=-1, **best_params).fit(X[tr], y[tr])

    dummy = DummyClassifier(strategy="most_frequent").fit(X[tr], y[tr])
    rf_proba = model.predict_proba(X[te])[:, 1]
    dm_proba = dummy.predict_proba(X[te])[:, 1]
    scores = pd.DataFrame({
        "Dummy(다수 클래스)": calc_metrics(y[te], dm_proba),
        "랜덤 포레스트": calc_metrics(y[te], rf_proba),
    }).T

    perm = permutation_importance(model, X[te], y[te], scoring="roc_auc",
                                  n_repeats=30, random_state=SEED, n_jobs=-1)
    importance = pd.DataFrame({
        "변수": [FEATURE_KR[f] for f in FEATURES],
        "트리 중요도": model.feature_importances_,
        "순열 중요도": perm.importances_mean,
        "순열 표준편차": perm.importances_std,
    })

    fpr, tpr, _ = roc_curve(y[te], rf_proba)
    return {
        "data": data, "model": model, "best_params": best_params,
        "cv_table": cv_table, "scores": scores, "importance": importance,
        "cm": confusion_matrix(y[te], (rf_proba >= 0.5).astype(int)),
        "fpr": fpr, "tpr": tpr,
        "n_train": len(tr), "n_test": len(te),
        "train_users": len(np.unique(groups[tr])),
        "test_users": len(np.unique(groups[te])),
        "test_prev": float(y[te].mean()),
    }

    st.markdown("""
- **트리 수**: 투표에 참여하는 결정 트리 개수예요. 많을수록 안정적이지만 일정 수준 넘으면 효과가 거의 없어요.
- **최대 깊이**: 트리가 질문을 몇 번까지 던질 수 있는지예요. **깊을수록 학습 데이터를 외우기 쉬워서(과적합)** 우리처럼 데이터가 적으면 얕게 제한해요.
- **리프 최소 표본**: 마지막 가지에 최소 몇 건이 있어야 하는지예요. 클수록 모델이 단순해져요.
- **튜닝 방법**: 학습 데이터 안에서 **사용자 단위 교차검증**으로 가장 좋은 조합을 골랐고, **시험 데이터는 마지막 평가에만** 썼어요.
  시험 데이터로 설정을 고르면 점수가 부풀려지기 때문이에요.
- 교차검증 AUC도 0.5 근처라면, 설정을 바꿔도 소용이 없다 = 입력 변수에 신호가 부족하다는 뜻이에요.
""")

# =========================================================
# 6. 직접 예측
# =========================================================
st.header("6️⃣ 직접 입력해서 예측해 보기")
c = st.columns(5)
steps = c[0].number_input("걸음 수", 0, 40000, 8000, 500)
very = c[1].number_input("매우 활발(분)", 0, 300, 20)
fairly = c[2].number_input("적당히 활발(분)", 0, 300, 20)
lightly = c[3].number_input("가벼운 활동(분)", 0, 600, 200)
sed = c[4].number_input("좌식(분)", 0, 1440, 800)

p = float(res["model"].predict_proba(np.array([[steps, very, fairly, lightly, sed]]))[0, 1])
st.metric("다음 날 수면 부족 확률(모델 점수)", f"{p:.1%}")
st.progress(min(max(p, 0.0), 1.0))
st.caption("⚠️ 이 값은 모델의 점수일 뿐이에요. 위의 AUC가 0.5 근처라면 실제 예측으로 믿으면 안 돼요. "
           "입력값을 바꿔도 확률이 거의 안 변한다면, 모델이 활동 변수에서 단서를 못 찾았다는 증거예요.")

# =========================================================
# 7. 종합 결론
# =========================================================
st.header("7️⃣ 종합 결론")
if auc < 0.55:
    st.error(
        f"랜덤 포레스트의 시험 AUC는 **{auc:.3f}**로 동전 던지기 수준이에요. "
        "이 데이터와 변수로는 다음 날 수면 부족을 예측할 만한 신호를 찾지 못했어요.")
elif auc < 0.65:
    st.warning(
        f"랜덤 포레스트의 시험 AUC는 **{auc:.3f}**로 약한 신호만 보여요. "
        "우연의 범위일 수 있으니 기준을 바꿔서 결과가 일관적인지 확인해야 해요.")
else:
    st.success(
        f"랜덤 포레스트의 시험 AUC는 **{auc:.3f}**로 어느 정도 구분 능력이 있어요. "
        "다만 사용자 수가 적으니 다른 분할과 다른 기준에서도 같은지 반드시 확인하세요.")

st.markdown("""
**⚠️ 결론을 쓸 때 주의할 점**
1. **"관계가 없다"가 아니라 "이 데이터·변수로는 신호를 못 찾았다"**라고 써야 정확해요. 사용자가 약 24명뿐이라 약한 관계는 못 잡을 수 있어요.
2. **하루 합계만 사용**했어요. 운동 시각 등은 알 수 없어요.
3. **좌식 시간에 수면이 섞였을 수 있어요.** 좌식 변수의 해석이 흐려질 수 있어요.
4. **시험 사용자가 적어서** 분할이 바뀌면 점수가 크게 달라져요. 메인의 반복 분할 방식으로도 확인하면 더 탄탄해요.
5. 사이드바에서 **수면 부족 기준을 6시간·7시간·8시간으로 바꿔** 결론이 그대로인지 확인해 보세요.
```
